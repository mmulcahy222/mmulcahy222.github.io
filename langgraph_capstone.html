"""25 • Retrieval + parallel specialists + review + durable human approval.

Four calls normally; at most five with one revision. Retrieval uses local text,
so there is no embedding API, vector database, web search, or extra account.
A Pydantic critic checks the draft; a human decides whether to keep it locally.
Observe: cited snippets, two specialist notes, bounded revision, and an interrupt.
Try: add a document and a test question; improve keyword retrieval or evaluation.
Run: python3 25_capstone_study_assistant.py
"""

from langgraph.graph import END, START, StateGraph

# A class groups related data or behavior. Inheriting from BaseModel gives Review
# Pydantic's validation and schema support. The critic must return these two fields
# instead of an unrestricted paragraph; validation checks shape/types, not truth.
# Defines the expected structure of the model's review.
class Review(BaseModel):
    # True asks for a correction; False means the critic found no required correction.
    needs_revision: bool
    # Adds guidance to the field's schema; this is not a default value.
    feedback: str = Field(description="One actionable issue, or 'No issues'")


# State is the shared dictionary passed between graph nodes. TypedDict describes
# its keys for tooling; it does not create a separate runtime object or validate
# values like BaseModel does. Fields are filled as the graph runs, so the initial
# input below supplies only question, notes, and revisions.
#
# Nodes return partial dictionaries containing only their updates. Most fields
# replace their previous value. The notes field instead uses a reducer: a function
# that combines updates. This matters because both parallel specialists write to
# notes; operator.add concatenates their lists rather than losing either result.
# Declares the graph's shared state schema.
class State(TypedDict):
    # The user's original study question.
    question: str
    # Matching local documents, each stored as an id/text dictionary.
    sources: list[dict]
    # Annotated attaches the list-concatenation reducer to the list[str] type.
    notes: Annotated[list[str], operator.add]
    # The current answer, or a clarification question if retrieval found nothing.
    draft: str
    # The critic's suggested improvement, if any.
    feedback: str
    # The critic's decision about whether the draft needs work.
    needs_revision: bool
    # Counts completed revisions so the graph can enforce a maximum of one.
    revisions: int
    # Records the human's final yes/no decision in the saved state.
    approved: bool


# Retrieval-augmented generation means finding reference material before asking
# the model to answer. This example uses keyword overlap in local text files.
# It does not understand synonyms, and repeated words do not increase the score.
# Accepts a string and returns a list of source dictionaries; -> is a return-type hint.
def retrieve_sources(question: str) -> list[dict]:
    # A docstring describes the function to readers and help().
    """A tiny lexical retriever; scores keywords, not meaning."""
    # A set holds unique common words to ignore.
    stop_words = {"the", "and", "how", "what", "does", "with", "that", "are"}
    # r"..." is a raw string used for the regex. [a-z]{3,} matches runs of at
    # least three ASCII letters. lower() makes matching case-insensitive here;
    # set() removes duplicates, and set subtraction removes the stop words.
    # Extracts the distinct search terms from the question.
    terms = set(re.findall(r"[a-z]{3,}", question.lower())) - stop_words
    # Starts an empty list that will hold (score, source) tuples.
    ranked = []
    
    # Path's / joins folders; glob finds text files, and sorted gives a stable scan order.
    for path in sorted((ROOT / "data" / "notes").glob("*.txt")):
        # Reads this entire file into a string.
        content = path.read_text()
        # & intersects the word sets; len counts distinct shared terms.
        score = len(terms & set(re.findall(r"[a-z]{3,}", content.lower())))
        # A nonzero integer is truthy, so only documents with a match are kept.
        if score:
            # Stores a tuple; path.stem is the filename without its .txt extension.
            ranked.append((score, {"id": path.stem, "text": content}))
    # lambda defines a small unnamed function that returns each item's sort key.
    # Python sorts tuples left to right: negative scores put larger scores first;
    # source IDs then break ties alphabetically so equal scores have stable order.
    # Sorts the existing list in place using score, then source ID.
    ranked.sort(key=lambda pair: (-pair[0], pair[1]["id"]))
    # A list comprehension unwraps the best two tuples; _ names the unused score and [:2] takes up to two items.
    return [source for _, source in ranked[:2]]


# Graph nodes receive the current state and return updates; they do not need to
# copy all existing fields into their return value. LangGraph merges the update.
# Adapts the retrieval helper into a graph node.
def retrieve(state):
    # Looks up the question, retrieves documents, and updates only sources.
    return {"sources": retrieve_sources(state["question"])}


# A routing function selects the next node(s); its return value is not a state
# update. A list of names schedules multiple branches in the same graph step.
# Defines the decision used after the retrieve node finishes.
def route_sources(
    # Receives the updated state, including the retrieved documents.
    state,
# Literal lists permitted strings; | means either that type or a list of strings.
) -> Literal["concepts", "examples", "no_sources"] | list[str]:
    # The inline if/else fans out when the list is nonempty, otherwise chooses the fallback.
    return ["concepts", "examples"] if state["sources"] else "no_sources"


# The two specialists use different prompts with the same cached model client.
# Each returns a one-item list; the notes reducer combines those lists. An f-string
# inserts expressions inside {...}; \n inserts a newline into the prompt text.
# The first specialist explains the underlying idea.
def concepts(state):
    # Gets the configured, cached chat-model client; the request happens in ask().
    model = get_model()
    # Inserts sources and question into the explanation request.
    prompt = f"Explain the core idea in 40 words using only these notes: {state['sources']}\nQuestion: {state['question']}"
    # ask invokes the model and extracts text; the label identifies this specialist's contribution.
    return {"notes": ["Concepts: " + ask(model, prompt)]}


# The second specialist proposes an exercise to practice the idea.
def examples(state):
    # Retrieves the same shared model configuration used by the other specialist.
    model = get_model()
    # Includes the reference material so the exercise relates to the question.
    prompt = f"Suggest a tiny practice exercise grounded in these notes: {state['sources']}\nQuestion: {state['question']}"
    # Calls the model and contributes one labeled note to the combined list.
    return {"notes": ["Practice: " + ask(model, prompt)]}


# The join edge below makes synthesis wait for both specialists. Adjacent string
# literals inside parentheses concatenate into one string without needing +.
# Word limits and citations here are prompt instructions, not checks in Python.
# Combines both specialists' notes and the source material into one answer.
def synthesize(state):
    # Gets the model client for the synthesis request.
    model = get_model()
    # Opens a parenthesized expression so the prompt can span several code lines.
    prompt = (
        # Requests the answer format and source references.
        f"Answer in 100 words with a practice task. Cite the provided source IDs in brackets. "
        # Asks the model to acknowledge gaps in the available evidence.
        f"Use only the supplied sources; say when they do not support a claim. "
        # Supplies the user's question and full retrieved documents.
        f"Question: {state['question']}\nSources: {state['sources']}\n"
        # Sorts a copy of the notes to give the prompt a stable order across parallel runs.
        f"Specialist notes: {sorted(state['notes'])}"
    # Ends the combined prompt expression; prompt is a single string.
    )
    # Generates the answer and stores it in the draft state field.
    return {"draft": ask(model, prompt)}


# Structured output asks the model for data matching a schema. Passing the Review
# Pydantic class makes the parsed result a Review instance whose fields match the
# state's feedback and needs_revision keys. The critique is still a model judgment.
# Evaluates the draft before choosing whether to revise it.
def review(state):
    # Gets the model client used to construct the critic.
    model = get_model()
    # Wraps the client with the Review schema; this line does not yet send the prompt.
    critic = model.with_structured_output(Review, method="json_schema")
    # Sends the review request and receives a parsed Review object.
    verdict = critic.invoke(
        # Defines what the critic should assess.
        f"Check whether this answer addresses the question and is supported by the sources. "
        # Provides the evidence and answer needed for that assessment.
        f"Question: {state['question']}\nSources: {state['sources']}\nAnswer: {state['draft']}"
    # Finishes the critic's model call.
    )
    # Converts the Pydantic object to a dictionary that updates feedback and needs_revision.
    return verdict.model_dump()


# Returns one of the two possible next-node names.
def route_review(state) -> Literal["revise", "human"]:
    # Both conditions must hold to revise; otherwise proceed to human approval.
    return "revise" if state["needs_revision"] and state["revisions"] < 1 else "human"


# This graph sends the revised answer directly to the human; it does not run the
# critic again. That topology, plus the routing guard, bounds the revision work.
# Makes the optional single correction requested by the critic.
def revise(state):
    # Gets the client for the revision request.
    model = get_model()
    # Supplies the criticism, original evidence, and draft to correct.
    prompt = f"Fix this answer in 100 words. Feedback: {state['feedback']}\nSources: {state['sources']}\nAnswer: {state['draft']}"
    # Replaces the draft and increments the revision counter in the same state update.
    return {"draft": ask(model, prompt), "revisions": state["revisions"] + 1}


# Handles questions that have no keyword overlap with the local documents.
def no_sources(state):
    # Still use the same model, but don't pretend there is retrieved evidence.
    # Gets the client for the fallback's only model call.
    model = get_model()
    # Begins a partial state update containing a clarification instead of a sourced answer.
    return {
        # Stores the model's clarification text under the usual draft key.
        "draft": ask(
            # Passes the configured model client as ask's first argument.
            model,
            # Requests a follow-up question and an explanation of the missing evidence.
            f"Ask one clarification question about: {state['question']}. Explain that the local notes have no matching sources.",
        # Completes the model call that supplies the draft value.
        )
    # Returns the draft update; this branch skips the specialists and critic.
    }


# interrupt is a durable pause, not terminal input. On the first visit, LangGraph
# suspends execution here and exposes the payload to the caller. The checkpointer
# preserves progress. Resuming with Command(resume=...) restarts this node from its
# beginning, and this interrupt call then returns the supplied resume value.
# Keep that restart behavior in mind before adding side effects before interrupt.
# Presents the draft for a human decision and records the answer on resume.
def human(state):
    # Pauses on the initial call; receives the resumed True/False value on the later call.
    approved = interrupt(
        # Builds the payload the caller can display while the graph is paused.
        {
            # Includes the answer or clarification for the human to read.
            "draft": state["draft"],
            # A list comprehension extracts source IDs without repeating the full document text.
            "sources": [s["id"] for s in state["sources"]],
            # Supplies the approval question shown with the draft.
            "question": "Keep this study note?",
        # Ends the interrupt payload dictionary.
        }
    # Once resumed, the interrupt's return value is assigned to approved.
    )
    # Requires an actual boolean rather than a string such as "yes".
    if not isinstance(approved, bool):
        # Rejects an incorrectly typed resume value with an explicit error.
        raise ValueError("Resume with a bool.")
    # Saves the human's decision as the final state update.
    return {"approved": approved}


# A graph describes execution using nodes (functions) and edges (what runs next).
# Building/compiling it prepares the workflow; it does not make model calls.
#
# With sources: retrieve -> concepts + examples -> synthesize -> review
#                       -> optional revise -> human -> END.
# Without sources: retrieve -> no_sources -> human -> END.
# Accepts the storage component that will save and restore graph progress.
def build_graph(checkpointer):
    # Creates a graph builder using the shared-state schema and its notes reducer.
    builder = StateGraph(State)
    # Iterates over (name, function) tuples, unpacking each into two variables.
    for name, node in [
        # Associates the routing name with the source-retrieval function, without calling it.
        ("retrieve", retrieve),
        # Registers the role that explains the core concept.
        ("concepts", concepts),
        # Registers the role that creates a practice exercise.
        ("examples", examples),
        # Registers the node that combines the parallel results.
        ("synthesize", synthesize),
        # Registers the structured-output critic.
        ("review", review),
        # Registers the optional correction step.
        ("revise", revise),
        # Registers the clarification path for unsuccessful retrieval.
        ("no_sources", no_sources),
        # Registers the node that interrupts for a human decision.
        ("human", human),
    # Finishes the list of node definitions and begins the loop body.
    ]:
        # Adds this function under its string name in the graph.
        builder.add_node(name, node)
    # Routes every new run from the virtual start marker to retrieval.
    builder.add_edge(START, "retrieve")
    # Adds routing whose destination depends on a function's result.
    builder.add_conditional_edges(
        # Runs route_sources after retrieve; the list declares possible destinations, not an execution order.
        "retrieve", route_sources, ["concepts", "examples", "no_sources"]
    # Completes the conditional routing definition.
    )
    # A single edge with a list of starting nodes is a join: synthesis waits for
    # both specialists to finish and for their notes updates to be combined.
    # Joins the two parallel branches before drafting the answer.
    builder.add_edge(["concepts", "examples"], "synthesize")
    # Always sends the initial synthesized answer to the critic.
    builder.add_edge("synthesize", "review")
    # Chooses revise or human; the router's Literal return hint declares its destinations.
    builder.add_conditional_edges("review", route_review)
    # Sends a corrected answer directly to the human without another review call.
    builder.add_edge("revise", "human")
    # Also asks for approval when the draft is only a clarification question.
    builder.add_edge("no_sources", "human")
    # Finishes after the resumed human node returns its decision.
    builder.add_edge("human", END)
    # Produces an executable graph connected to persistent checkpoint storage.
    return builder.compile(checkpointer=checkpointer)


# A thread ID identifies one saved run's history. Reusing it with the same SQLite
# database lets a new process find the paused run. The database stores checkpoint
# state (including the draft) whether the human eventually approves or rejects it;
# approval records a decision rather than controlling whether any data is saved.
# Coordinates command-line options, persistent storage, graph execution, and terminal input.
def main():
    # Imports Python's standard command-line parser locally for this entry point.
    import argparse

    # Uses the module's opening docstring as the --help description.
    parser = argparse.ArgumentParser(description=__doc__)
    # Defines an optional question argument for a new run.
    parser.add_argument(
        # Supplies this default when no --question value is given.
        "--question", default="How do checkpoints and interrupts work together?"
    # Finishes the question option definition.
    )
    # Defines an option for continuing an existing paused run.
    parser.add_argument(
        # The user passes --resume followed by the previously printed thread ID.
        "--resume",
        # Uses a readable placeholder in the command's help text.
        metavar="THREAD_ID",
        # Explains the option in --help output.
        help="Resume an existing paused run after a restart",
    # Finishes the resume option definition.
    )
    # Parses command-line values into attributes such as args.question and args.resume.
    args = parser.parse_args()
    # or keeps a supplied nonempty ID; otherwise it creates and stringifies a new UUID.
    thread_id = args.resume or str(uuid4())
    # Identifies the checkpoint history and caps graph execution steps, not Python call-stack depth.
    config = {"configurable": {"thread_id": thread_id}, "recursion_limit": 16}
    # with is a context manager: it opens the SQLite saver for this block and
    # closes its connection on exit, including when an exception leaves the block.
    # data_dir() creates the configured data folder if needed; / joins the filename,
    # and str() converts the Path object into the connection string the saver needs.
    # Opens the on-disk checkpoint database for both new and resumed runs.
    with SqliteSaver.from_conn_string(str(data_dir() / "capstone.sqlite")) as saver:
        # Compiles the graph using the open saver so interrupts survive a process restart.
        graph = build_graph(saver)
        # Follows this branch when the user supplied an existing thread ID.
        if args.resume:
            # Loads that thread's latest saved state and pending-node information.
            snapshot = graph.get_state(config)
            # snapshot.next is a tuple of node names scheduled to run next.
            # ("human",) is a one-item tuple; the comma distinguishes it from a
            # parenthesized string. This check requires saved state awaiting human.
            # Rejects missing threads and threads not positioned at the approval node.
            if not snapshot.values or snapshot.next != ("human",):
                # Ends the command with a useful error instead of resuming the wrong state.
                raise SystemExit("That thread is not waiting for human approval.")
            # Shows the previously generated draft without rerunning its model calls.
            print(snapshot.values["draft"])
        # Starts a fresh run when no --resume option was supplied.
        else:
            # Runs the workflow until the human node interrupts execution.
            paused = graph.invoke(
                # Seeds the question, empty reducer list, and counter, with this thread's configuration.
                {"question": args.question, "notes": [], "revisions": 0}, config
            # Returns the current state together with interrupt information.
            )
            # Reads the first interrupt object's payload: the draft, source IDs, and approval question.
            print(paused["__interrupt__"][0].value)
        # Displays recovery instructions before waiting for terminal input.
        print(
            # Includes the exact ID needed to reopen this paused history.
            f"Thread: {thread_id}\nYou can exit now and resume later with --resume {thread_id}"
        # Finishes printing the two-line recovery message.
        )
        # Reads text, trims whitespace, lowercases it, and produces True only for y (including uppercase Y).
        approved = input("Keep note? [y/N] ").strip().lower() == "y"
        # Resumes the same thread; the boolean becomes interrupt's return value and the graph finishes.
        result = graph.invoke(Command(resume=approved), config)
        # Reports the recorded decision after execution reaches END.
        print(
            # Labels the value being printed.
            "Approved:",
            # Reads the boolean stored by the human node.
            result["approved"],
            # Explains where the result lives; rejecting it does not delete the checkpoint.
            "(saved in the local checkpoint; nothing published)",
        # Completes the status message; print separates these arguments with spaces.
        )
        # Returns the final state to callers; leaving the with block also closes the database connection.
        return result


# Runs this block only when launched as a script, not when imported by another module.
if __name__ == "__main__":
    # noqa: F405  # Wraps this module's functions with tracing and includes their source code in trace output.
    decorate_all_functions(show_code=True)
    # Starts the command-line study-assistant workflow.
    main()
